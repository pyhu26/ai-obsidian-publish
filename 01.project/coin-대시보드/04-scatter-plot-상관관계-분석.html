📝 구현 과정

1. TrendScatterPlot 컴포넌트

src/components/TrendScatterPlot.tsx

import { useMemo } from 'react';
import {
  ScatterChart, Scatter, XAxis, YAxis, CartesianGrid,
  Tooltip, Legend, ResponsiveContainer, Line,
} from 'recharts';
import { calculateLinearRegression, getTrendLinePoints } from '../utils/linearRegression';
import type { MarketData } from '../types';

interface Props {
  data: MarketData[];
  selectedMarkets?: string[];
  onMarketClick?: (marketName: string) => void;
}

export function TrendScatterPlot({ data, selectedMarkets = [], onMarketClick }: Props) {
  const { scatterData, trendLineData } = useMemo(() => {
    // Scatter 데이터 변환
    const scatterData = data.map((item) => ({
      x: item.trend_weight,
      y: item.last_high_price,
      market_name: item.market_name,
      isSelected: selectedMarkets.includes(item.market_name),
    }));

    // 선형 회귀 계산
    const regressionData = data.map((item) => ({
      x: item.trend_weight,
      y: item.last_high_price,
    }));

    const { slope, intercept } = calculateLinearRegression(regressionData);
    const trendLineData = getTrendLinePoints(regressionData, slope, intercept);

    return { scatterData, trendLineData };
  }, [data, selectedMarkets]);

  const handleClick = (data: any) => {
    if (onMarketClick && data && data.market_name) {
      onMarketClick(data.market_name);
    }
  };

  return (
    <ResponsiveContainer width="100%" height={500}>
      <ScatterChart
        margin={{ top: 20, right: 30, left: 20, bottom: 20 }}
      >
        <CartesianGrid strokeDasharray="3 3" />
        <XAxis
          type="number"
          dataKey="x"
          name="트렌드 가중치"
          label={{ value: '트렌드 가중치', position: 'insideBottom', offset: -10 }}
        />
        <YAxis
          type="number"
          dataKey="y"
          name="최고가"
          label={{ value: '최고가 (원)', angle: -90, position: 'insideLeft' }}
        />
        <Tooltip
          cursor={{ strokeDasharray: '3 3' }}
          content={({ payload }) => {
            if (!payload || payload.length === 0) return null;
            const data = payload[0].payload;
            return (
              <div className="bg-white p-3 border border-gray-300 rounded shadow-lg">
                <p className="font-bold text-gray-900">{data.market_name}</p>
                <p className="text-sm text-gray-600">
                  트렌드: {data.x.toFixed(6)}
                </p>
                <p className="text-sm text-gray-600">
                  가격: {data.y.toLocaleString()}원
                </p>
              </div>
            );
          }}
        />
        <Legend />

        {/* 추세선 */}
        <Line
          type="monotone"
          data={trendLineData}
          dataKey="y"
          stroke="#ef4444"
          strokeWidth={2}
          strokeDasharray="5 5"
          dot={false}
          name="추세선"
        />

        {/* Scatter 점들 */}
        <Scatter
          name="마켓"
          data={scatterData}
          fill="#3b82f6"
          onClick={handleClick}
          shape={(props: any) => {
            const { cx, cy, payload } = props;
            const isSelected = payload.isSelected;
            return (
              <circle
                cx={cx}
                cy={cy}
                r={isSelected ? 8 : 5}
                fill={isSelected ? '#ef4444' : '#3b82f6'}
                stroke={isSelected ? '#dc2626' : '#2563eb'}
                strokeWidth={2}
                style={{ cursor: 'pointer' }}
              />
            );
          }}
        />
      </ScatterChart>
    </ResponsiveContainer>
  );
}

핵심 포인트

  1. Custom Shape: 선택된 마켓은 크기/색상 변경
  2. onClick 핸들러: 클릭 시 부모 컴포넌트에 알림
  3. Custom Tooltip: 마켓명과 정확한 값 표시

2. 인터랙션 구현

Dashboard에서 상태 관리:

// Dashboard.tsx
export function Dashboard() {
  const { data } = useMarketData();
  const [selectedMarket, setSelectedMarket] = useState<string>('');

  const handleMarketClick = (marketName: string) => {
    setSelectedMarket((prev) => (prev === marketName ? '' : marketName));
  };

  return (
    <TrendScatterPlot
      data={data}
      selectedMarkets={selectedMarket ? [selectedMarket] : []}
      onMarketClick={handleMarketClick}
    />
  );
}

상태 관리 로직

  • 같은 마켓 클릭 → 선택 해제 (토글 방식)
  • 다른 마켓 클릭 → 새로운 마켓 선택

3. Custom Tooltip 디자인

<Tooltip
  content={({ payload }) => {
    if (!payload || payload.length === 0) return null;
    const data = payload[0].payload;

    return (
      <div className="bg-white p-3 border border-gray-300 rounded shadow-lg">
        <p className="font-bold text-gray-900">
          {data.market_name}
        </p>
        <p className="text-sm text-gray-600">
          트렌드: {data.x.toFixed(6)}
        </p>
        <p className="text-sm text-gray-600">
          가격: {data.y.toLocaleString()}원
        </p>
      </div>
    );
  }}
/>

스타일 특징

  • shadow-lg: 깊은 그림자로 부각
  • toLocaleString(): 가격 천 단위 구분
  • toFixed(6): 트렌드 가중치 소수점 6자리

클릭한 마켓 (빨간색 ●)

상관관계 분석

추세선 해석

  • 기울기가 양수: 트렌드 ↑ → 가격 ↑ (양의 상관관계)
  • 기울기가 음수: 트렌드 ↑ → 가격 ↓ (음의 상관관계)
  • 기울기가 0에 가까움: 상관관계 없음

실제 데이터 분석

// 예시 결과
slope: 15000000  // 양의 상관관계
// 해석: 트렌드 가중치가 0.1 증가하면 가격이 약 150만원 증가

💡 배운 점

1. ScatterChart의 특징

// Bar/Line Chart와 달리 XAxis, YAxis 모두 숫자 타입
<XAxis type="number" dataKey="x" />
<YAxis type="number" dataKey="y" />

// Line Chart에서는
<XAxis dataKey="time" />  // 카테고리
<YAxis />                  // 숫자

2. Custom Shape 활용

shape={(props) => {
  const { cx, cy, payload } = props;
  // cx, cy: 중심 좌표
  // payload: 데이터

  return (
    <circle
      cx={cx}
      cy={cy}
      r={payload.isSelected ? 8 : 5}  // 조건부 크기
      fill={payload.isSelected ? 'red' : 'blue'}
    />
  );
}}

3. 이벤트 핸들링

// Recharts는 클릭된 데이터를 자동으로 전달
<Scatter onClick={(data) => console.log(data)} />

// data 구조:
{
  x: 0.5,
  y: 1000000,
  market_name: "KRW-BTC",
  payload: { ... }  // 전체 데이터
}

4. 데이터 시각화의 가치

  • 숫자만 보면 알 수 없던 패턴 발견
  • 이상치(outlier) 쉽게 파악
  • 클러스터링 확인

🐛 트러블슈팅

문제 1: Y축 값이 너무 커서 읽기 어려움

원인: 가격 범위가 1,000원 ~ 100,000,000원으로 매우 큼

시도한 해결책:

// 로그 스케일 적용
<YAxis scale="log" domain={['auto', 'auto']} />

결과: 데이터 분포가 고르게 표시됨

문제 2: 클릭이 제대로 동작하지 않음

원인: SVG 요소의 z-index 문제

해결: Scatter를 Line 뒤에 선언

<Line ... />      {/* 추세선 먼저 */}
<Scatter ... />   {/* Scatter 나중에 (위에 그려짐) */}

문제 3: Tooltip이 화면 밖으로 벗어남

원인: Recharts 기본 Tooltip 위치 계산

해결: wrapperStyle로 조정

<Tooltip
  wrapperStyle={{ zIndex: 1000 }}
  allowEscapeViewBox={{ x: true, y: true }}
/>
