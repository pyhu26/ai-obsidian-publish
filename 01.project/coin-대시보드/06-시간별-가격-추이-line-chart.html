1. 시계열 데이터 타입 추가

src/types/index.ts

export interface MarketData {
  market_name: string;
  trend_weight: number;
  last_high_price: number;
  high_prices: number[] | null;      // 가격 배열
  timestamp_array: number[] | null;   // 시간 배열 (인덱스)
  analysis_time: string | null;
  created_at: string;
}

// 추세선 데이터
export interface TrendLine {
  marketName: string;
  slope: number;
  intercept: number;
  points: { time: number; price: number }[];
}

데이터 구조

  • high_prices: [100, 105, 110, 108, ...] (가격 배열)
  • timestamp_array: [0, 1, 2, 3, ...] (시간 인덱스)

2. PriceLineChart 컴포넌트

src/components/PriceLineChart.tsx

import { useMemo } from 'react';
import {
  LineChart, Line, XAxis, YAxis, CartesianGrid,
  Tooltip, Legend, ResponsiveContainer,
} from 'recharts';
import { calculateLinearRegression, getTrendLinePoints } from '../utils/linearRegression';
import type { MarketData } from '../types';

interface Props {
  data: MarketData[];
  selectedMarket: string;
}

export function PriceLineChart({ data, selectedMarket }: Props) {
  const chartData = useMemo(() => {
    if (!selectedMarket) return null;

    const market = data.find((d) => d.market_name === selectedMarket);
    if (!market || !market.high_prices || !market.timestamp_array) {
      return null;
    }

    // 시계열 데이터 변환
    const priceData = market.high_prices.map((price, index) => ({
      time: market.timestamp_array![index],
      price: price,
      formattedTime: `T-${market.high_prices!.length - index}`,  // T-10, T-9, ...
    }));

    // 추세선 계산
    const regressionData = priceData.map((d) => ({ x: d.time, y: d.price }));
    const { slope, intercept } = calculateLinearRegression(regressionData);
    const trendLine = getTrendLinePoints(regressionData, slope, intercept);

    // 추세선 데이터 병합
    return priceData.map((d, index) => ({
      ...d,
      trendLine: trendLine[index].y,
    }));
  }, [data, selectedMarket]);

  if (!chartData) {
    return (
      <div className="flex items-center justify-center h-64 text-gray-500">
        {selectedMarket
          ? '해당 마켓의 시계열 데이터가 없습니다.'
          : '마켓을 선택하세요.'}
      </div>
    );
  }

  return (
    <ResponsiveContainer width="100%" height={400}>
      <LineChart
        data={chartData}
        margin={{ top: 20, right: 30, left: 20, bottom: 20 }}
      >
        <CartesianGrid strokeDasharray="3 3" />
        <XAxis
          dataKey="formattedTime"
          label={{ value: '시간', position: 'insideBottom', offset: -10 }}
        />
        <YAxis
          label={{ value: '가격 (원)', angle: -90, position: 'insideLeft' }}
        />
        <Tooltip
          formatter={(value: number) => [value.toLocaleString() + '원', '가격']}
        />
        <Legend />

        {/* 실제 가격 라인 */}
        <Line
          type="monotone"
          dataKey="price"
          stroke="#3b82f6"
          strokeWidth={2}
          dot={{ r: 3 }}
          name={selectedMarket}
        />

        {/* 추세선 */}
        <Line
          type="monotone"
          dataKey="trendLine"
          stroke="#ef4444"
          strokeWidth={2}
          strokeDasharray="5 5"
          dot={false}
          name="추세선"
        />
      </LineChart>
    </ResponsiveContainer>
  );
}

핵심 구현

  1. 시간 포맷팅: T-10, T-9, ... (현재부터 10 시점 전, 9 시점 전)
  2. 빈 상태 처리: 선택된 마켓이 없거나 데이터가 없을 때 안내 메시지
  3. 추세선 병합: 실제 가격과 추세선을 같은 차트에 표시

3. 차트 간 마켓 선택 동기화

src/components/Dashboard.tsx

export function Dashboard() {
  const { data } = useMarketData();
  const [selectedMarket, setSelectedMarket] = useState<string>('');

  const handleMarketClick = (marketName: string) => {
    // 토글 방식: 같은 마켓 클릭 시 선택 해제
    setSelectedMarket((prev) => (prev === marketName ? '' : marketName));
  };

  return (
    <div>
      {/* Bar Chart */}
      <TrendBarChart
        data={data}
        selectedMarkets={selectedMarket ? [selectedMarket] : []}
        onMarketClick={handleMarketClick}
      />

      {/* Scatter Plot */}
      <TrendScatterPlot
        data={data}
        selectedMarkets={selectedMarket ? [selectedMarket] : []}
        onMarketClick={handleMarketClick}
      />

      {/* Line Chart */}
      <PriceLineChart
        data={data}
        selectedMarket={selectedMarket}
      />
    </div>
  );
}

동기화 메커니즘

  • 모든 차트가 같은 selectedMarket 상태 공유
  • Bar Chart나 Scatter Plot에서 마켓 클릭 → Line Chart 자동 업데이트
  • 상태 끌어올리기(Lifting State Up) 패턴

추세 분석

추세선 기울기

slope > 0  // 상승 추세 📈
slope < 0  // 하락 추세 📉
slope ≈ 0  // 횡보 추세 →

예시

// KRW-BTC
slope: 5000000    // 시간당 500만원 상승
intercept: 50000000
// → 강한 상승 추세

💡 배운 점

1. 시계열 데이터 처리

// ❌ 나쁜 예: 데이터 중복
const chartData = data.map(d => ({
  time: d.time,
  price: d.price,
  trendLine: calculateTrend(d),  // 매번 계산
}));

// ✅ 좋은 예: 한 번 계산 후 매핑
const { slope, intercept } = calculateLinearRegression(data);
const trendLine = getTrendLinePoints(data, slope, intercept);
const chartData = data.map((d, i) => ({
  ...d,
  trendLine: trendLine[i].y,  // 미리 계산된 값 사용
}));

2. 조건부 렌더링 전략

// Early return으로 깔끔하게
if (!chartData) {
  return <EmptyState />;
}

return <Chart data={chartData} />;

🐛 트러블슈팅

문제 1: 추세선이 실제 가격과 너무 다름

원인: 이상치(outlier)의 영향

해결: 이상치 제거 또는 가중 회귀

// 간단한 이상치 제거
const cleanedData = data.filter(d => {
  const mean = avg(data);
  const std = standardDeviation(data);
  return Math.abs(d.price - mean) < 2 * std;
});

문제 2: X축 라벨이 겹침

원인: 데이터 포인트가 너무 많음

해결: 간격 조절

<XAxis
  dataKey="formattedTime"
  interval={Math.floor(chartData.length / 10)}  // 10개만 표시
/>

문제 3: 선택 해제가 안 됨

원인: 토글 로직 누락

해결: 같은 마켓 클릭 시 빈 문자열로 설정

setSelectedMarket((prev) => (prev === marketName ? '' : marketName));
