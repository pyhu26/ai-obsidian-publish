1. useDarkMode 훅 생성

src/hooks/useDarkMode.ts

import { useEffect, useState } from 'react';

export function useDarkMode() {
  const [isDark, setIsDark] = useState(() => {
    // 1순위: 로컬스토리지에 저장된 사용자 설정
    const saved = localStorage.getItem('darkMode');
    if (saved !== null) {
      return saved === 'true';
    }

    // 2순위: 시스템 설정 확인
    return window.matchMedia('(prefers-color-scheme: dark)').matches;
  });

  useEffect(() => {
    const root = window.document.documentElement;

    // HTML 루트 요소에 'dark' 클래스 추가/제거
    if (isDark) {
      root.classList.add('dark');
    } else {
      root.classList.remove('dark');
    }

    // LocalStorage에 설정 저장
    localStorage.setItem('darkMode', isDark.toString());
  }, [isDark]);

  const toggleDarkMode = () => setIsDark(!isDark);

  return { isDark, toggleDarkMode };
}

다크 모드 우선순위

  1. 사용자가 설정한 값 (LocalStorage)
  2. 시스템 설정 (prefers-color-scheme)
  3. 기본값 (라이트 모드)

5. LocalStorage vs SessionStorage

// LocalStorage: 브라우저 종료해도 유지
localStorage.setItem('darkMode', 'true');

// SessionStorage: 탭 닫으면 삭제

다크 모드는 LocalStorage 사용 권장

  • 다른 탭에서도 동일한 설정 유지

🐛 트러블슈팅

문제 1: 다크 모드 깜빡임 (Flash of Unstyled Content)

원인: React가 마운트되기 전 기본 라이트 모드로 표시

해결: 인라인 스크립트로 초기 클래스 설정

<!-- index.html -->
<script>
  const isDark = localStorage.getItem('darkMode') === 'true' ||
    (!localStorage.getItem('darkMode') &&
     window.matchMedia('(prefers-color-scheme: dark)').matches);
  if (isDark) {
    document.documentElement.classList.add('dark');
  }
</script>

해결: 최소 터치 타겟 크기 보장

<button className="min-h-[44px] min-w-[44px] px-4 py-2">

원인: Recharts 기본 색상이 어두운 배경에 안 보임

해결: 다크 모드에 맞는 색상 사용

// 라이트 모드: 진한 색
stroke={isDark ? "#60A5FA" : "#2563EB"}  // blue-400 vs blue-600

<Line stroke="var(--chart-color)" />

:root {
  --chart-color: #2563EB;
}

.dark {
  --chart-color: #60A5FA;
}
